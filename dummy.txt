<script>
/* ===================== DATA ===================== */
const TEAM_COLORS = ['--t1','--t2','--t3','--t4','--t5','--t6'];
const DEFAULT_NAMES = ['Chennai Kings','Mumbai Titans','Bengaluru Royals','Kolkata Riders','Hyderabad Sunrisers','Punjab Warriors'];

const FIRST_NAMES = ['Rohan','Vikram','Arjun','Suresh','Karan','Aditya','Rahul','Manoj','Siddharth','Nikhil','Deepak','Aakash','Ishaan','Yuvraj','Harshad','Tanay','Rajat','Gaurav','Vivek','Aman','Shivam','Kunal','Anirudh','Farhan','Sameer','Naveen','Pranav','Rishabh','Devendra','Chirag','Mohit','Sagar','Abhishek','Kartik','Zaheer','Imran','Parth','Ravi','Sandeep','Tarun','Varun','Yash','Ajay','Ankit','Bhuvan','Digvijay','Girish','Harsh','Jatin','Lakshay'];
const LAST_NAMES = ['Sharma','Verma','Patel','Iyer','Reddy','Nair','Chauhan','Malhotra','Kapoor','Bhatt','Menon','Rathore','Deshmukh','Joshi','Pillai','Shetty','Gill','Kohli','Yadav','Mishra','Saxena','Trivedi','Bose','Chatterjee','Rao','Naidu','Pandey','Dubey','Khan','Ansari','Chandra','Bakshi','Ghosh','Dutta','Sinha','Tiwari','Rawat','Bhardwaj','Sood','Oberoi'];
const ROLES = [
  {key:'BAT', label:'Batsman', weight:0.32},
  {key:'BOWL', label:'Bowler', weight:0.32},
  {key:'AR', label:'All-Rounder', weight:0.20},
  {key:'WK', label:'Wicketkeeper', weight:0.16}
];

function pick(arr){ return arr[Math.floor(Math.random()*arr.length)]; }
function weightedRole(){
  const r = Math.random(); let acc=0;
  for(const role of ROLES){ acc += role.weight; if(r<=acc) return role; }
  return ROLES[0];
}
function randInt(min,max){ return Math.floor(Math.random()*(max-min+1))+min; }

function generatePlayerPool(n){
  const pool = []; const usedNames = new Set();
  for(let i=0;i<n;i++){
    let name;
    do { name = pick(FIRST_NAMES) + ' ' + pick(LAST_NAMES); } while(usedNames.has(name));
    usedNames.add(name);
    const role = weightedRole();
    // Skew a small % of players to be "marquee" (higher rating & base price)
    const isMarquee = Math.random() < 0.12;
    const rating = isMarquee ? randInt(88,99) : randInt(52,90);
    const basePrice = isMarquee ? [2,3,4][randInt(0,2)] : [0.3,0.5,0.75,1,1.5][randInt(0,4)];
    pool.push({ id:i, name, role: role.key, roleLabel: role.label, rating, basePrice });
  }
  // Shuffle & sort loosely so marquee names appear spread through the auction, not all first
  for(let i=pool.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); [pool[i],pool[j]]=[pool[j],pool[i]]; }
  return pool;
}

/* ===================== STATE ===================== */
let state = null;

function fmtCr(v){
  return '₹' + (Math.round(v*100)/100).toString().replace(/\.0$/,'') + ' Cr';
}

/* ===================== SETUP SCREEN ===================== */
const teamSetupGrid = document.getElementById('teamSetupGrid');
function buildTeamSetupGrid(){
  teamSetupGrid.innerHTML = '';
  for(let i=0;i<6;i++){
    const row = document.createElement('div');
    row.className = 'team-setup-row';
    row.innerHTML = `
      <div class="team-swatch" style="background:var(${TEAM_COLORS[i]})"></div>
      <input type="text" id="teamName${i}" value="${DEFAULT_NAMES[i]}" placeholder="Team ${i+1} name">
    `;
    teamSetupGrid.appendChild(row);
  }
}
buildTeamSetupGrid();

document.getElementById('startBtn').addEventListener('click', () => {
  const purse = parseFloat(document.getElementById('purseInput').value) || 100;
  const squadSize = parseInt(document.getElementById('squadSizeInput').value) || 11;
  const teams = [];
  for(let i=0;i<6;i++){
    const nameInput = document.getElementById('teamName'+i);
    teams.push({
      id:i,
      name: nameInput.value.trim() || DEFAULT_NAMES[i],
      color: TEAM_COLORS[i],
      purse: purse,
      startPurse: purse,
      squad: [],
      squadSize: squadSize,
      passedThisLot: false
    });
  }
  const poolSize = 50;
  state = {
    teams,
    pool: generatePlayerPool(poolSize),
    poolSize,
    currentIndex: 0,
    currentPlayer: null,
    currentBid: 0,
    leaderId: null,
    bidStep: 0.5,
    soldCount: 0,
    unsoldCount: 0
  };
  document.getElementById('screen-setup').classList.add('hidden');
  document.getElementById('screen-auction').classList.remove('hidden');
  nextLot();
});

/* ===================== AUCTION SCREEN ===================== */
const teamBidGrid = document.getElementById('teamBidGrid');
const teamsPanel = document.getElementById('teamsPanel');
const soldStamp = document.getElementById('soldStamp');

function teamsStillNeedingPlayers(){
  return state.teams.filter(t => t.squad.length < t.squadSize);
}

function nextLot(){
  soldStamp.classList.remove('show','unsold-stamp');
  soldStamp.textContent = 'SOLD';

  const needy = teamsStillNeedingPlayers();
  if(state.currentIndex >= state.pool.length || needy.length === 0){
    return showResults();
  }

  const player = state.pool[state.currentIndex];
  state.currentPlayer = player;
  state.currentBid = player.basePrice;
  state.leaderId = null;
  state.teams.forEach(t => t.passedThisLot = false);

  document.getElementById('lotEyebrow').textContent = `LOT ${state.currentIndex+1} OF ${state.pool.length}`;
  document.getElementById('pName').textContent = player.name;
  document.getElementById('pRole').textContent = player.roleLabel;
  document.getElementById('pBase').textContent = 'Base ' + fmtCr(player.basePrice);
  document.getElementById('pRating').textContent = player.rating;
  document.getElementById('bidAmount').textContent = fmtCr(state.currentBid);
  document.getElementById('bidLeader').textContent = 'No bids yet — base price';
  document.getElementById('soldBtn').disabled = true;
  document.getElementById('lotCounter').textContent = `${state.soldCount} sold · ${state.unsoldCount} unsold · ${state.pool.length - state.currentIndex - 1} remaining`;

  renderTeamBidGrid();
  renderTeamsPanel();
}

function canTeamBid(team){
  if(team.squad.length >= team.squadSize) return false;
  if(team.passedThisLot) return false;
  if(team.id === state.leaderId) return false;
  const nextBid = state.currentBid + state.bidStep;
  // must be able to afford this bid AND still afford minimum base price for remaining slots after this buy
  const remainingSlotsAfter = team.squadSize - team.squad.length - 1;
  const reserveNeeded = remainingSlotsAfter * 0.3; // minimum reserve so they can still fill the squad
  return (team.purse - nextBid) >= reserveNeeded;
}

function renderTeamBidGrid(){
  teamBidGrid.innerHTML = '';
  state.teams.forEach(team => {
    const btn = document.createElement('button');
    btn.className = 'team-bid-btn' + (team.id === state.leaderId ? ' leading' : '');
    btn.style.borderColor = team.id === state.leaderId ? 'var(--gold)' : '';
    const disabled = !canTeamBid(team);
    btn.disabled = disabled;
    btn.innerHTML = `${team.name}<br><span style="color:var(${team.color});font-weight:700;">${fmtCr(team.purse)}</span>`;
    btn.addEventListener('click', () => placeBid(team.id));
    teamBidGrid.appendChild(btn);
  });
}

function placeBid(teamId){
  const team = state.teams.find(t => t.id === teamId);
  state.currentBid += state.bidStep;
  state.leaderId = teamId;
  document.getElementById('bidAmount').textContent = fmtCr(state.currentBid);
  document.getElementById('bidLeader').innerHTML = `Leading: <b style="color:var(${team.color})">${team.name}</b>`;
  document.getElementById('soldBtn').disabled = false;
  renderTeamBidGrid();
}

document.getElementById('soldBtn').addEventListener('click', () => {
  if(state.leaderId === null) return;
  const team = state.teams.find(t => t.id === state.leaderId);
  team.purse -= state.currentBid;
  team.squad.push({ ...state.currentPlayer, price: state.currentBid });
  state.soldCount++;

  soldStamp.textContent = 'SOLD';
  soldStamp.classList.remove('unsold-stamp');
  soldStamp.classList.add('show');

  renderTeamsPanel();
  setTimeout(() => {
    state.currentIndex++;
    nextLot();
  }, 900);
});

document.getElementById('unsoldBtn').addEventListener('click', () => {
  state.unsoldCount++;
  soldStamp.textContent = 'UNSOLD';
  soldStamp.classList.add('show','unsold-stamp');
  setTimeout(() => {
    state.currentIndex++;
    nextLot();
  }, 700);
});

function renderTeamsPanel(){
  teamsPanel.innerHTML = '';
  state.teams.forEach(team => {
    const full = team.squad.length >= team.squadSize;
    const card = document.createElement('div');
    card.className = 'team-card' + (full ? ' out' : '');
    card.style.setProperty('--tc', `var(${team.color})`);
    const pct = Math.min(100, (team.squad.length/team.squadSize)*100);
    card.innerHTML = `
      <div class="team-card-top">
        <span class="tname">${team.name}</span>
        <span class="tpurse">${fmtCr(team.purse)}</span>
      </div>
      <div class="team-card-bottom">
        <span>${team.squad.length}/${team.squadSize} players</span>
        <span>${full ? 'Squad full' : 'Building squad'}</span>
      </div>
      <div class="progress-bar"><div class="progress-fill" style="width:${pct}%;background:var(${team.color})"></div></div>
    `;
    teamsPanel.appendChild(card);
  });
}

/* ===================== RESULTS SCREEN ===================== */
function showResults(){
  document.getElementById('screen-auction').classList.add('hidden');
  document.getElementById('screen-results').classList.remove('hidden');

  const ranked = [...state.teams].sort((a,b) => {
    const sumA = a.squad.reduce((s,p)=>s+p.rating,0);
    const sumB = b.squad.reduce((s,p)=>s+p.rating,0);
    return sumB - sumA;
  });

  const winner = ranked[0];
  const winnerTotal = winner.squad.reduce((s,p)=>s+p.rating,0);
  document.getElementById('winnerName').textContent = winner.name;
  document.getElementById('winnerSub').textContent = `Total squad rating: ${winnerTotal} across ${winner.squad.length} players`;

  const grid = document.getElementById('resultsGrid');
  grid.innerHTML = '';
  ranked.forEach((team, idx) => {
    const total = team.squad.reduce((s,p)=>s+p.rating,0);
    const card = document.createElement('div');
    card.className = 'squad-card';
    card.style.setProperty('--tc', `var(${team.color})`);
    const sortedSquad = [...team.squad].sort((a,b)=>b.rating-a.rating);
    card.innerHTML = `
      <h3>${idx===0 ? '🏆 ' : ''}${team.name}</h3>
      <div class="squad-total">Total rating: ${total}</div>
      <ul class="squad-list">
        ${sortedSquad.map(p => `<li><span class="nm">${p.name} <span style="color:var(--muted);font-size:11px;">(${p.role})</span></span><span>${p.rating}</span></li>`).join('')}
        ${team.squad.length === 0 ? '<li style="color:var(--muted);">No players bought</li>' : ''}
      </ul>
    `;
    grid.appendChild(card);
  });
}
